def traffic_step(times, paces, pass_success=False, blocking=True):
    # Two-car illustrative intervention, with equal times broken by index.
    if len(times) != 2 or len(paces) != 2 or not all(math.isfinite(x) for x in [*times,*paces]) or any(x <= 0 for x in paces):
        raise ValueError('two finite times and positive paces required')
    if type(pass_success) is not bool or type(blocking) is not bool:
        raise ValueError('boolean intervention flags required')
    order = sorted(range(2),key=lambda i:(times[i],i))
    leader,follower = order
    resolved = list(paces)
    gap = times[follower]-times[leader]
    if blocking and gap < 1 and not pass_success:
        resolved[follower] = max(resolved[follower],resolved[leader]+dirty_air(gap))
    return [times[i]+resolved[i] for i in range(2)]
