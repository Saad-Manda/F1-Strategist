import hashlib

def stable_seed(master,label):
    return int.from_bytes(hashlib.sha256(f'{master}:{label}'.encode()).digest()[:4],'big')

def schedule(scenarios, opponents, master_seed):
    for xs in (scenarios,opponents):
        if not xs or any(not isinstance(x,str) or not x for x in xs) or len(set(xs)) != len(xs):
            raise ValueError('distinct nonempty IDs required')
    if type(master_seed) is not int or master_seed < 0:
        raise ValueError('nonnegative integer master seed required')
    return [dict(scenario_id=s,opponent_id=o,scenario_seed=stable_seed(master_seed,'scenario:'+s),
                 opponent_seed=stable_seed(master_seed,'opponent:'+o),exploration=0)
            for s in scenarios for o in opponents]

def experience(vector_envs, vector_steps):
    if type(vector_envs) is not int or vector_envs < 1 or type(vector_steps) is not int or vector_steps < 0:
        raise ValueError('positive environment count and nonnegative steps required')
    return vector_envs*vector_steps

Recap and self-check

Does a valid schedule certify a strong learner?

No. It establishes the comparison conditions; measured behavior under those conditions is separate evidence.
